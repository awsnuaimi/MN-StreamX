cmake_minimum_required(VERSION 3.22.1)
project(mnx_captions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS mnx_compile=2 mnx_link=1)
set(CMAKE_JOB_POOL_COMPILE mnx_compile)
set(CMAKE_JOB_POOL_LINK mnx_link)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag GGML_NATIVE GGML_OPENMP GGML_OPENMP_FETCH GGML_BLAS GGML_LLAMAFILE GGML_CCACHE
        GGML_CPU_REPACK GGML_SSE42 GGML_AVX GGML_AVX2 GGML_BMI2 GGML_FMA GGML_F16C
        WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_CURL)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
if(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(GGML_CPU_ARM_ARCH "armv7-a" CACHE STRING "" FORCE)
endif()
if(NOT WHISPER_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(whisper_source
        URL https://codeload.github.com/ggml-org/whisper.cpp/tar.gz/refs/tags/v1.9.5
        URL_HASH SHA256=ff1a9053feb509ff9d7729703355541ae9690073a6b1c40eb692c962e0dc1720)
    FetchContent_MakeAvailable(whisper_source)
else()
    add_subdirectory(${WHISPER_SOURCE_DIR} whisper_source)
endif()
include(${CMAKE_CURRENT_SOURCE_DIR}/WhisperWindowPatch.cmake)
if(WHISPER_SOURCE_DIR)
    mnx_patch_whisper_window(${WHISPER_SOURCE_DIR})
else()
    mnx_patch_whisper_window(${whisper_source_SOURCE_DIR})
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
if(NOT SENTENCEPIECE_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(sentencepiece_source
        URL https://codeload.github.com/google/sentencepiece/tar.gz/refs/tags/v0.2.0
        URL_HASH SHA256=9970f0a0afee1648890293321665e5b2efa04eaec9f1671fcf8048f456f5bb86)
    FetchContent_GetProperties(sentencepiece_source)
    if(NOT sentencepiece_source_POPULATED)
        FetchContent_Populate(sentencepiece_source)
    endif()
    set(SENTENCEPIECE_SOURCE_DIR ${sentencepiece_source_SOURCE_DIR})
endif()
add_subdirectory(${SENTENCEPIECE_SOURCE_DIR} sentencepiece_source EXCLUDE_FROM_ALL)
add_library(mnx-whisper SHARED WhisperBridge.cpp SentencePieceBridge.cpp)
target_include_directories(mnx-whisper PRIVATE ${SENTENCEPIECE_SOURCE_DIR}/src)
target_link_libraries(mnx-whisper PRIVATE sentencepiece-static)
target_link_libraries(mnx-whisper PRIVATE whisper)
if(ANDROID)
    target_link_libraries(mnx-whisper PRIVATE log)
    target_link_options(mnx-whisper PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_include_directories(mnx-whisper PRIVATE ${JNI_INCLUDE_DIRS})
endif()
